cmake_minimum_required(VERSION 3.16)
project(serialbench-c C CXX)

include(FetchContent)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(LibXml2)
find_package(LibXslt)
find_package(Jansson)
find_package(LibYAML)

# tris cores: explicit -DTRIS_LIB_DIR/-DTRIS_INC_DIR or per-lib overrides
set(TRIS_LIB_DIR "${TRIS_LIB_DIR}" CACHE PATH "dir containing built tris .so/.dylib")
set(TRIS_INC_DIR "${TRIS_INC_DIR}" CACHE PATH "dir containing tris headers")
find_library(LEPTRIS_LIB NAMES libleptris libtaurus HINTS "${TRIS_LIB_DIR}" "${LEPTRIS_DIR}" ENV LEPTRIS_DIR)
find_library(YEPTRIS_LIB NAMES libyeptris HINTS "${TRIS_LIB_DIR}" "${YEPTRIS_DIR}" ENV YEPTRIS_DIR)
find_library(TEPTRIS_LIB NAMES libteptris HINTS "${TRIS_LIB_DIR}" "${TEPTRIS_DIR}" ENV TEPTRIS_DIR)
set(LEPTRIS_INC "${LEPTRIS_INC}" CACHE PATH "leptris include dirs")
set(YEPTRIS_INC "${YEPTRIS_INC}" CACHE PATH "yeptris include dirs")
set(TEPTRIS_INC "${TEPTRIS_INC}" CACHE PATH "teptris include dirs")

find_package(PkgConfig QUIET)
pkg_check_modules(JSON_C QUIET IMPORTED_TARGET json-c)
find_package(nlohmann_json 3.2.0 QUIET)
# tomlc17: header + source from external dir or FetchContent
if(EXISTS "${TOMLC17_DIR}/src/tomlc17.h")
  set(TOMLC17_SRC "${TOMLC17_DIR}/src/tomlc17.c")
  set(TOMLC17_INC "${TOMLC17_DIR}/src")
endif()
FetchContent_Declare(ryml GIT_REPOSITORY https://github.com/biojppm/rapidyaml GIT_TAG v0.7.2 GIT_SHALLOW TRUE)
option(FETCH_RYML "fetch rapidyaml from source" ON)
if(FETCH_RYML AND NOT HAVE_SYSTEM_RYML)
  FetchContent_MakeAvailable(ryml)
endif()
FetchContent_Declare(simdjson GIT_REPOSITORY https://github.com/simdjson/simdjson GIT_TAG v3.13.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(simdjson)
FetchContent_Declare(pugixml GIT_REPOSITORY https://github.com/zeux/pugixml GIT_TAG v1.16 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(pugixml)
FetchContent_Declare(libcbor GIT_REPOSITORY https://github.com/PJK/libcbor GIT_TAG v0.14.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(libcbor)
FetchContent_Declare(tinycbor GIT_REPOSITORY https://github.com/intel/tinycbor GIT_TAG v0.6.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(tinycbor)

add_executable(serialbench-c src/bench.c src/bench_extra.c src/bench_cpp.cpp)
if(ryml_FOUND OR TARGET ryml)
  target_compile_definitions(serialbench-c PRIVATE HAVE_RYML=1)
  target_link_libraries(serialbench-c PRIVATE ryml::ryml)
endif()
if(nlohmann_json_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_NLOHMANN=1)
  target_link_libraries(serialbench-c PRIVATE nlohmann_json::nlohmann_json)
endif()
if(JSON_C_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_JSON_C=1)
  target_link_libraries(serialbench-c PRIVATE PkgConfig::JSON_C)
endif()
if(TOMLC17_SRC)
  target_compile_definitions(serialbench-c PRIVATE HAVE_TOMLC17=1)
  target_include_directories(serialbench-c PRIVATE ${TOMLC17_INC})
  target_sources(serialbench-c PRIVATE ${TOMLC17_SRC})
endif()
if(TARGET simdjson::simdjson)
  target_compile_definitions(serialbench-c PRIVATE HAVE_SIMDJSON=1)
  target_link_libraries(serialbench-c PRIVATE simdjson::simdjson)
endif()
if(TARGET pugixml::pugixml)
  target_compile_definitions(serialbench-c PRIVATE HAVE_PUGIXML=1)
  target_link_libraries(serialbench-c PRIVATE pugixml::pugixml)
endif()
if(TARGET libcbor::libcbor)
  target_compile_definitions(serialbench-c PRIVATE HAVE_LIBCBOR=1)
  target_link_libraries(serialbench-c PRIVATE libcbor::libcbor)
endif()
# tinycbor ships no CMakeLists (v0.6.1): compile the decode path directly
if(EXISTS "${tinycbor_SOURCE_DIR}/src/cborparser.c")
  target_compile_definitions(serialbench-c PRIVATE HAVE_TINYCBOR=1
    TINYCBOR_CBOR_H="${tinycbor_SOURCE_DIR}/src/cbor.h")
  target_sources(serialbench-c PRIVATE
    "${tinycbor_SOURCE_DIR}/src/cborparser.c"
    "${tinycbor_SOURCE_DIR}/src/cborparser_dup_string.c"
    "${tinycbor_SOURCE_DIR}/src/cborparser_float.c"
    "${tinycbor_SOURCE_DIR}/src/cborerrorstrings.c"
    "${tinycbor_SOURCE_DIR}/src/cborvalidation.c")
endif()
target_compile_definitions(serialbench-c PRIVATE HAVE_EXTRA_BENCH=1)

if(LIBXML2_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_LIBXML2=1)
  target_include_directories(serialbench-c PRIVATE ${LIBXML2_INCLUDE_DIRS})
  target_link_libraries(serialbench-c PRIVATE ${LIBXML2_LIBRARIES})
  message(STATUS "libxml2: found")
else()
  message(WARNING "libxml2: NOT found")
endif()

if(LIBXSLT_FOUND AND LIBXML2_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_LIBXSLT=1)
  target_include_directories(serialbench-c PRIVATE ${LIBXSLT_INCLUDE_DIRS})
  target_link_libraries(serialbench-c PRIVATE ${LIBXSLT_LIBRARIES})
  message(STATUS "libxslt: found")
else()
  message(STATUS "libxslt: skipped")
endif()


if(PkgConfig_FOUND)
  pkg_check_modules(JANSSON QUIET IMPORTED_TARGET jansson)
  pkg_check_modules(LIBYAML QUIET IMPORTED_TARGET yaml-0.1)
endif()

if(JANSSON_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_JANSSON=1)
  target_link_libraries(serialbench-c PRIVATE PkgConfig::JANSSON)
  message(STATUS "jansson: found")
else()
  message(STATUS "jansson: skipped")
endif()

if(LIBYAML_FOUND)
  target_compile_definitions(serialbench-c PRIVATE HAVE_LIBYAML=1)
  target_link_libraries(serialbench-c PRIVATE PkgConfig::LIBYAML)
  message(STATUS "libyaml: found")
else()
  message(STATUS "libyaml: skipped")
endif()

find_library(UTF8PROC_LIB NAMES utf8proc libutf8proc)
find_package(Iconv QUIET)
if(LEPTRIS_LIB AND LEPTRIS_INC)
  target_compile_definitions(serialbench-c PRIVATE HAVE_LEPTRIS=1)
  target_include_directories(serialbench-c PRIVATE ${LEPTRIS_INC})
  target_link_libraries(serialbench-c PRIVATE ${LEPTRIS_LIB})
  if(UTF8PROC_LIB)
    target_link_libraries(serialbench-c PRIVATE ${UTF8PROC_LIB})
    message(STATUS "leptris utf8proc: ${UTF8PROC_LIB}")
  endif()
  if(Iconv_FOUND)
    target_link_libraries(serialbench-c PRIVATE Iconv::Iconv)
    message(STATUS "leptris iconv: found")
  endif()
  message(STATUS "leptris: ${LEPTRIS_LIB}")
else()
  message(STATUS "leptris: skipped")
endif()

if(YEPTRIS_LIB AND YEPTRIS_INC)
  target_compile_definitions(serialbench-c PRIVATE HAVE_YEPTRIS=1)
  target_include_directories(serialbench-c PRIVATE ${YEPTRIS_INC})
  target_link_libraries(serialbench-c PRIVATE ${YEPTRIS_LIB})
  message(STATUS "yeptris: ${YEPTRIS_LIB}")
else()
  message(STATUS "yeptris: skipped")
endif()

if(TEPTRIS_LIB AND TEPTRIS_INC)
  target_compile_definitions(serialbench-c PRIVATE HAVE_TEPTRIS=1)
  target_include_directories(serialbench-c PRIVATE ${TEPTRIS_INC})
  target_link_libraries(serialbench-c PRIVATE ${TEPTRIS_LIB})
  message(STATUS "teptris: ${TEPTRIS_LIB}")
else()
  message(STATUS "teptris: skipped")
endif()
